\subsection{Fast I/O}
Reads ints only ($10^6+$). Redirect with \texttt{./a.out < input.txt}. \textbf{Note:} Do NOT mix \texttt{readInt()} with \texttt{cin} or \texttt{cout}.
\begin{minted}{c++}
inline char gc() {
  static char buf[1 << 16];
  static size_t bc, be;
  if (bc >= be) {
    buf[0] = 0, bc = 0;
    be = fread(buf, 1, sizeof(buf), stdin);
  }
  return buf[bc++];
}
int readInt() {
  int a, c;
  while ((a = gc()) < 40) if (a == 0) return 0;
  bool minus = (a == '-');
  if (minus) a = gc();
  unsigned res = a - 48;
  while ((c = gc()) >= 48) res = res * 10 + (c - 48);
  return minus ? -(ll)res : (int)res;
}
\end{minted}
\subsection{Debug}
Supports primitives, \texttt{string}, \texttt{pair}, containers (\texttt{vector}, \texttt{set}, \texttt{map}, matrices), and \texttt{queue} (see note for \texttt{stack}/\texttt{pq}). Disabled with \texttt{-DONLINE\_JUDGE}.
\begin{minted}{c++}
#ifndef ONLINE_JUDGE
#define dbg(...) cerr << "[" #__VA_ARGS__ "]:", pr(__VA_ARGS__)
string to_string(char c) { return string(1, c); }
string to_string(bool b) { return b ? "T" : "F"; }
string to_string(const char* s) { return s; }
string to_string(char* s) { return s; }
string to_string(string s) { return s; }
template <class A, class B>
string to_string(pair<A, B> p);
template <class A>
string to_string(A v) {
  string r = "{"; int f = 0;
  for (auto& x : v)
    r += (f++ ? "," : "") + to_string(x);
  return r + "}";
}
template <class A, class B>
string to_string(pair<A, B> p) {
  return "(" + to_string(p.first) + ","
             + to_string(p.second) + ")";
}
template <class T, class... C>
string to_string(queue<T, C...> q) {
  string r = "{"; int f = 0;
  for (; q.size(); q.pop())
    r += (f++ ? "," : "") + to_string(q.front());
  return r + "}";
}
// For stack / priority_queue: replace queue with stack / priority_queue & .front() with .top()
void pr() { cerr << "\n"; }
template <class H, class... T>
void pr(H h, T... t) {
  cerr << " " << to_string(h); pr(t...);
}
#else
#define dbg(...)
#endif
\end{minted}
