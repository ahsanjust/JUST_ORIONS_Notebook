\subsection{Fast I/O}
Reads ints only (no strings/floats). Use for large input ($10^6+$). Redirect using \texttt{./a.out < input.txt}.
\textbf{Note:} If you use \texttt{readInt()}, do NOT use \texttt{cin} or \texttt{cout} anywhere else in the code.
\begin{minted}{c++}
inline char gc() {
  static char buf[1 << 16];
  static size_t bc, be;
  if (bc >= be) {
    buf[0] = 0, bc = 0;
    be = fread(buf, 1, sizeof(buf), stdin);
  }
  return buf[bc++];
}
int readInt() {
  int a, c;
  while ((a = gc()) < 40) if (a == 0) return 0;
  bool minus = (a == '-');
  if (minus) a = gc();
  unsigned res = a - 48;
  while ((c = gc()) >= 48) res = res * 10 + (c - 48);
  return minus ? -(long long)res : (int)res;
}
// ~5x faster than cin/scanf
\end{minted}
\subsection{Debug}
\texttt{dbg(x1,x2,...)} prints \texttt{x1=value1 || x2=value2 || ...} to stderr. Works with integers, strings, pairs, containers (vector, set, map). Disabled with \texttt{-DONLINE\_JUDGE}.
\begin{minted}{c++}
// In main.cpp:
#ifndef ONLINE_JUDGE
#include "trace.cpp"
#else
#define dbg(...)
#endif

// ---- paste below into trace.cpp ----
#ifndef ONLINE_JUDGE
#define dbg(...) cerr<<__LINE__<<": ", \
  __dbg(#__VA_ARGS__,__VA_ARGS__), cerr<<"\n"

// primitive types (int,ll,char,string,etc.)
template<class T> void __p(T x) { cerr<<x; }
// pair<T1,T2>
template<class T,class U> void __p(pair<T,U> x)
{ cerr<<"("<<x.first<<","<<x.second<<")"; }
// vector<T>
template<class T> void __p(vector<T> x)
{ cerr<<"{";int _=0;for(auto y:x){if(_++)cerr<<",";__p(y);}cerr<<"}"; }
// set<T>
template<class T> void __p(set<T> x)
{ cerr<<"{";int _=0;for(auto y:x){if(_++)cerr<<",";__p(y);}cerr<<"}"; }
// map<K,V>
template<class T,class U> void __p(map<T,U> x)
{ cerr<<"{";int _=0;for(auto[k,v]:x){if(_++)cerr<<",";__p(k);cerr<<":";__p(v);}cerr<<"}"; }
// stack<T>
template<class T> void __p(stack<T> s)
{ vector<T> v;while(v.size()||s.size())v.push_back(s.top()),s.pop();
  reverse(v.begin(),v.end());__p(v); }
// queue<T>
template<class T> void __p(queue<T> q)
{ vector<T> v;while(q.size())v.push_back(q.front()),q.pop();__p(v); }
// priority_queue<T>
template<class T> void __p(priority_queue<T> q)
{ vector<T> v;while(q.size())v.push_back(q.top()),q.pop();__p(v); }

void __dbg(const char*){}
template<class T,class...U>
void __dbg(const char*s,T x,U...t){
  int i=0;while(s[i]&&s[i]!=',')i++;
  cerr.write(s,i)<<"=";__p(x);
  if constexpr(sizeof...(t)) cerr<<" || ",__dbg(s+i+1,t...);
}
#else
#define dbg(...)
#endif
\end{minted}
